1    import java.util.Date;
2    import java.util.HashMap;
3    import java.util.Map;
4    
5    
6    public class GPSCollar extends Sensor{
7        // boundary of the zone: [minLat, maxLat, minLon, maxLon]
8        private double minLat, maxLat, minLon, maxLon;
9    
10       private PositionRecord[] record;
11       private HashMap<Date, PositionRecord[]> history;
12   
13       // ==================== CONSTRUCTOR ====================
14       public GPSCollar(int code, int zoneCode,
15                        double minLat, double maxLat,
16                        double minLon, double maxLon) {
17           super(code, zoneCode);
18           this.minLat = minLat;
19           this.maxLat = maxLat;
20           this.minLon = minLon;
21           this.maxLon = maxLon;
22           this.record = new PositionRecord[24];
23           this.history = new HashMap<>();
24       }
25   
26       //  METHODS
27       public void addPosition(int hour, Position p) {
28           if (hour < 0 || hour > 23) return;
29           boolean outOfBoundary = isOutOfBoundary(p);
30           record[hour] = new PositionRecord(p, outOfBoundary);
31           if (outOfBoundary) {
32               triggerBoundaryAlert(p, hour);
33           }
34       }
35   
36       public boolean isOutOfBoundary(Position p) {
37           return p.getLatitude()  < minLat || p.getLatitude()  > maxLat ||
38                   p.getLongitude() < minLon || p.getLongitude() > maxLon;
39       }
40   
41   
42   
43       private void triggerBoundaryAlert(Position p, int hour) {
44           System.out.println("ALERT! Collar " + getCode() +
45                   " left zone at hour " + hour + " | " + p);
46       }
47   
48       public void saveDailyRecord(Date date) {
49           for (int i = 0; i < 24; i++) {
50               if (record[i] == null) {
51                   System.out.println("Day not complete, hour " + i + " missing.");
52                   return;
53               }
54           }
55           PositionRecord[] copy = new PositionRecord[24];
56           System.arraycopy(record, 0, copy, 0, 24);
57           history.put(date, copy);
58           record = new PositionRecord[24];
59           System.out.println("GPS day saved successfully!");
60       }
61   
62       public PositionRecord[] getRecord() { return record; }
63       public HashMap<Date, PositionRecord[]> getHistory() { return history; }
64   
65       @Override
66       public void browseHistory(Date startDate, Date endDate) throws IllegalArgumentException {
67   
68           // ========== EXCEPTIONS ==========
69           if (startDate == null || endDate == null) {
70               throw new IllegalArgumentException("Dates cannot be null.");
71           }
72           if (startDate.after(endDate)) {
73               throw new IllegalArgumentException("Start date cannot be after end date.");
74           }
75   
76           // ========== BROWSE ==========
77           boolean foundAny = false;
78   
79           for (Map.Entry<Date, PositionRecord[]> entry : history.entrySet()) {
80               Date date = entry.getKey();
81   
82               if (!date.before(startDate) && !date.after(endDate)) {
83                   foundAny = true;
84                   System.out.println("==========================================");
85                   System.out.println("Date: " + date);
86                   System.out.println("==========================================");
87   
88                   PositionRecord[] dayRecord = entry.getValue();
89                   for (int hour = 0; hour < 24; hour++) {
90                       if (dayRecord[hour] != null) {
91                           System.out.println("  Hour " + String.format("%02d", hour) + ":00"
92                                   + " | Lat: " + dayRecord[hour].getPosition().getLatitude()
93                                   + " Lon: " + dayRecord[hour].getPosition().getLongitude()
94                                   + " | Out of boundary: " + dayRecord[hour].isOutOfBoundary());
95                       } else {
96                           System.out.println("  Hour " + String.format("%02d", hour) + ":00"
97                                   + " | No position recorded.");
98                       }
99                   }
100              }
101          }
102  
103          if (!foundAny) {
104              System.out.println("No records found between " + startDate + " and " + endDate);
105          }
106      }

108  }
