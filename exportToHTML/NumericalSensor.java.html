1    import java.util.Date;
2    import java.util.HashMap;
3    import java.util.Map;
4    
5    public abstract class NumericalSensor extends Sensor {
6        public enum Unit {
7            CELSIUS, PERCENTAGE, MM, PH, MG_PER_L, STEPS_PER_MIN, MG_PER_KG
8        }
9        // attributes
10       private Unit unit;
11       private Reading[] record;       // 24 slots, one per hour
12       private double minThreshold;
13       private double maxThreshold;
14       private char period;           // 'h' for hourly
15       private HashMap<Date, Reading[]> history;  // key: date, value: 24h record
16   
17       // Methods
18       public NumericalSensor(int  code, int  zoneCode, Unit unit,
19                              double minThreshold, double maxThreshold) {
20           super(code, zoneCode);
21           this.unit = unit;
22           this.minThreshold = minThreshold;
23           this.maxThreshold = maxThreshold;
24           this.record = new Reading[24];
25           this.period = 'h';
26           this.history = new HashMap<>();
27       }
28       // add reading each hour
29       public void addReading(int hour, double value) {
30           if (hour < 0 || hour > 23) return;
31           boolean outOfRange = isOutOfRange(value);
32           record[hour] = new Reading(value, outOfRange);
33           if (outOfRange) {
34               triggerAlert(value, hour);
35           }
36       }
37       // check if the recorded value at determined hour
38       public boolean isOutOfRange(double value) {
39           return value < minThreshold || value > maxThreshold;
40       }
41       private void triggerAlert(double value, int hour) {
42           System.out.println("⚠ ALERT! Sensor " + getCode() +
43                   " | Value: " + value + " " + unit +
44                   " at hour " + hour +
45                   " | Threshold: [" + minThreshold + ", " + maxThreshold + "]");
46       }
47       // save the day redord in the history
48       public void saveDailyRecord(Date date) {
49           // check if all 24 slots are filled
50           for (int i = 0; i < 24; i++) {
51               if (record[i] == null) {
52                   System.out.println("Day not complete yet, hour " + i + " is missing.");
53                   return; // don't save, day is not finished
54               }
55           }
56           // all 24 readings are present → save and reset
57           Reading[] copy = new Reading[24];
58           System.arraycopy(record, 0, copy, 0, 24);
59           history.put(date, copy);
60           record = new Reading[24]; // reset for new day
61           System.out.println("Day saved successfully!");
62       }
63       // ==================== GETTERS & SETTERS ====================
64       public Unit getUnit() { return unit; }
65   
66       public Reading[] getRecord() { return record; }
67   
68       public double getMinThreshold() { return minThreshold; }
69   
70       public double getMaxThreshold() { return maxThreshold; }
71   
72       public char getPeriod() { return period; }
73   
74       public HashMap<Date,Reading[]> getHistory() { return history; }
75   
76       public void setMinThreshold(double min) { this.minThreshold = min; }
77   
78       public void setMaxThreshold(double max) { this.maxThreshold = max; }
79   
80       @Override
81       public void browseHistory(Date startDate, Date endDate) throws IllegalArgumentException {
82   
83           // ========== EXCEPTIONS ==========
84           if (startDate == null || endDate == null) {
85               throw new IllegalArgumentException("Dates cannot be null.");
86           }
87           if (startDate.after(endDate)) {
88               throw new IllegalArgumentException("Start date cannot be after end date.");
89           }
90   
91           // ========== BROWSE ==========
92           boolean foundAny = false;
93   
94           for (Map.Entry<Date, Reading[]> entry : history.entrySet()) {
95               Date date = entry.getKey();
96   
97               // check if date is within range (inclusive)
98               if (!date.before(startDate) && !date.after(endDate)) {
99                   foundAny = true;
100                  System.out.println("==========================================");
101                  System.out.println("Date: " + date);
102                  System.out.println("==========================================");
103  
104                  Reading[] dayRecord = entry.getValue();
105                  for (int hour = 0; hour < 24; hour++) {
106                      if (dayRecord[hour] != null) {
107                          System.out.println("  Hour " + String.format("%02d", hour) + ":00"
108                                  + " | Value: " + dayRecord[hour].getValue()
109                                  + " " + getUnit()
110                                  + " | Out of range: " + dayRecord[hour].isOutOfRange());
111                      } else {
112                          System.out.println("  Hour " + String.format("%02d", hour) + ":00"
113                                  + " | No reading recorded.");
114                      }
115                  }
116              }
117          }
118  
119          if (!foundAny) {
120              System.out.println("No records found between " + startDate + " and " + endDate);
121          }
122      }
123  }
